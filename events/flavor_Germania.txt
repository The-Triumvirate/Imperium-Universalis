country_event = {
	id = germania.1
	title = "germania.1.t"
	desc = "germania.1.d"
	picture = city

	trigger = {
		tag = CHT
		1760 = {
			is_empty = yes
		}
		any_owned_province = {
			any_neighbor_province = { province_id = 1760 }
		}
	}

	mean_time_to_happen = {
		months = 90
	}

	option = {
		name = "germania.1.a"
		custom_tooltip = germania.1.tt
		hidden_effect = {
			1760 = {
				create_colony = 1000
			}
		}
	}
	option = {
		name = "germania.1.b"
		add_stability = -1
	}
}

					tribal_land_of = ROOT

		}
		modifier = {
			factor = 0.9
			mil = 6
		}
		modifier = {
			factor = 0.75
			ruler_has_personality = expansionist_personality
		}
		modifier = {
			factor = 0.5
			is_year = 700
		}
	}
	option = {
		name = "germania.4.a"
		if = {

				any_owned_province = {
					culture_group = gaulish_group
					OR = {

						region = germani_cisrhenani

				}
				change_culture = ROOT
				change_religion = ROOT
			}
		}
		else = {
			random_province = {
				limit = {
					tribal_land_of = ROOT
					culture_group = gaulish_group
					OR = {
						region = germania_region

						region = hercynia_region

					}
				}

	id = germania.5
	title = "germania.5.t"
	desc = "germania.5.d"
	picture = village

	trigger = {
		OR = {
			culture_group = germanic_group
			culture_group = gaulish_group
		}
		OR = {
			any_owned_province = {
				is_core = ROOT
				culture_group = north_atlantic_group
			}
			any_province = {
				AND = {
					tribal_land_of = ROOT
					culture_group = north_atlantic_group
				}
			}
		}
	}

	mean_time_to_happen = {
		months = 180
		modifier = {
			factor = 0.9
			mil = 4
		}
		modifier = {
			factor = 0.9
			mil = 5
		}
		modifier = {
			factor = 0.9
			mil = 6
		}
		modifier = {
			factor = 0.9
			stability = 1
		}
		modifier = {
			factor = 0.9
			stability = 2
		}
		modifier = {
			factor = 0.9
			stability = 3
		}
		modifier = {
			factor = 0.75
			ruler_has_personality = expansionist_personality
		}
		modifier = {
			factor = 0.5
			is_year = 300
		}
	}
	option = {
		name = "germania.4.a"
		if = {
			limit = {
				any_owned_province = {
					culture_group = north_atlantic_group
					is_core = ROOT
				}
			}
			random_owned_province = {
				limit = {
					culture_group = north_atlantic_group

					change_culture = ROOT
					change_religion = ROOT
				}
			}
		}
		else = {
			random_province = {
				limit = {
					tribal_land_of = ROOT

				}
				if = {
					limit = {
						ROOT = {
							culture_group = germanic_group
						}
					}
					change_culture = istvaeonic
					change_religion = ROOT
				}
				else = {
					change_culture = ROOT
					change_religion = ROOT
				}
			}
