country_event = {
	id = ai_driving.3
	title = "ai_driving.3.t"
	desc = "ai_driving.3.d"
	picture = maniple

	fire_only_once = yes
	hidden = yes

	trigger = {
		tag = ROM
		has_global_flag = week_passed_start
	}

	mean_time_to_happen = {
		days = 1
	}

	option = {
		name = "ai_driving.3.a"
		ai_chance = { factor = 100 }
		latium_area = { add_permanent_claim = ROM }
		66 = { add_permanent_claim = ROM }
		5644 = { add_permanent_claim = ROM }
		54 = { add_permanent_claim = ROM }
		if = {
			limit = {
				ROM = { AI = yes }
				NOT = { has_global_flag = no_italy_RR }
			}
			socii_clear_check = yes
			latium_area = { add_permanent_claim = ROM }
			66 = { add_permanent_claim = ROM }
			5644 = { add_permanent_claim = ROM }
			54 = { add_permanent_claim = ROM }
			if = { #for free mana later
				limit = {
					has_reform = roman_kingdom_reform
				}
				set_country_flag = started_in_kingdom_flag
			}
			if = {
				limit = { NOT = { total_development = 150 } }
				add_country_modifier = {
					name = rom_small_buff
					duration = -1
				}
				set_country_flag = rome_is_small_at_start
				add_manpower = 8
				add_treasury = 100
			}
			53 = {
				add_province_modifier = {
					name = rom_ai_def_buff
					duration = -1
				}
			}
			#Kill Socii so rome AI doesn't implode
			every_subject_country = {
				if = {
					limit = {
						is_subject_of_type = socii_subject
					}
					ROM = { inherit = PREV }
				}
			}
			socii_clear_check = yes
			every_known_country = {
				limit = {
					capital_scope = {
						region = italian_region
					}
					NOT = { tag = ROM }
					ai = yes
				}
				add_country_modifier = {
					name = ita_ai_rome_diplo_debuff
					duration = -1
				}
			}
			#Yearly repeating event for AI rome to complete missions and further railroad them
			country_event = {
				days = 365
				id = ai_driving.31
			}
		}
		if = {
			limit = {
				VOL = { AI = yes }
			}
			VOL = {
				add_country_modifier = {
					name = vol_buff
					duration = 18250
				}
			}
		}
	}
}

country_event = { #check ai rome every year
	id = ai_driving.31
	title = "ai_driving.31.t"
	desc = "ai_driving.31.d"
	picture = maniple
	is_triggered_only = yes
	hidden = yes

	trigger = {
		tag = ROM
		ai = yes
	}

	option = {
		country_event = {
			days = 365
			id = ai_driving.31
		}
		if = {
			limit = {
				had_country_flag = {
					flag = rome_is_small_at_start
					days = 18250
				}
				NOT = { num_of_cities = 10 }
				government = republic
				any_known_country = {
					primary_culture = latin
				}
			}
			country_event = {
				id = ai_driving.33
			}
		}
		if = {
			limit = {
				has_country_flag = started_in_kingdom_flag
				has_reform = res_publica_reform
			}
			clr_country_flag = started_in_kingdom_flag
			add_country_modifier = {
				name = rom_ai_mana_buff
				duration = 18250
			}
		}
		if = {
			limit = {
				OR = {
					total_development = 400
					CAR = { total_development = 400 }
					any_known_country = {
						culture_group = hellenic_group
						total_development = 300
					}
				}
				OR = {
					government = republic
					CAR = { total_development = 400 }
				}
				has_country_modifier = rom_small_buff
			}
			remove_country_modifier = rom_small_buff
			add_country_modifier = {
				name = rom_medium_buff
				duration = -1
			}
			country_event = {
				id = ai_driving.35
			}
		}
		if = { #give free subjects sometimes... AI too scared to invade if they dont have allies (subjects)
			limit = {
				has_country_modifier = rom_medium_buff
				NOT = { num_of_non_tributary_subjects = 8 }
				num_of_free_diplomatic_relations = 1
			}
			random = {
				chance = 20
				random_neighbor_country = {
					limit = {
						NOT = { total_development = 100 }
						is_subject = no
						NOT = { OR = { historical_rival_with = ROM ROM = { historical_rival_with = PREV } } }
						NOT = { has_opinion_modifier = { who = THIS modifier = insulted } }
						is_at_war = no
					}
				}
				country_event = {
					id = ai_driving.36
				}
			}
		}
		if = { #give socii system modifier
			limit = {
				num_of_cities = 30
			}
			country_event = {
				id = ai_driving.32
			}
		}
		rom_ai_harmonize_randomly_effect = yes
		ai_rome_forum_mission_effect = yes
		ai_rome_conquer_first_colonia_effect = yes
		ai_rome_build_first_colonia_effect = yes
		ai_rome_conquer_other_latins_effect = yes
		ai_rome_conquer_via_appia_effect = yes
		ai_rome_build_via_appia_effect = yes
	}
}
