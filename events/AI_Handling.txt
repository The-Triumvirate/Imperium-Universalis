country_event = {
	id = ai_handling.4
	title = "NONE"
	desc = "NONE"
	picture = "NONE"
	is_triggered_only = yes
	hidden = yes

	trigger = {
		has_global_flag = ai_building_scripts_on
		is_in_deficit = no
		is_at_war = no
		treasury = 350
		NOT = { num_of_loans = 1 }
		NOT = { primitives = yes }
		NOT = { government = native }
	}

	immediate = {
		if = {
			limit = {
				ai_can_build_fort = yes
				capital_scope = {
					ai_has_any_fort_trigger = no
				}
			}
			capital_scope = {
				ai_build_forts_effect = yes
			}
		}
		else_if = {
			limit = {
				ai_can_build_fort = yes
			}
			random_owned_province = {
				limit = {
					OR = {
						best_fort_province_trigger = yes
						AND = {
							OR = {
								high_terrain_trigger = yes
								forest_terrain_trigger = yes
								has_terrain = marsh
							}
							OR = {
								AND = {
									NOT = {
										OR = {
											ai_is_building_fort_trigger = yes
											ai_has_any_fort_trigger = yes
											has_influencing_fort = yes
										}
									}
									all_neighbor_province = {
										NOT = {
											OR = {
												ai_is_building_fort_trigger = yes
												ai_has_any_fort_trigger = yes
												has_influencing_fort = yes
											}
										}
									}
								}
							}
						}
					}
				}
				ai_build_forts_effect = yes
			}
		}
	}

	after = {
		ROOT = {
			add_country_modifier = {
				name = ai_build_fort_cooldown
				duration = 412
			}
		}
	}
	option = {}
}

#BUILDING BUILDER
country_event = {
	id = ai_handling.5
	title = "NONE"
	desc = "NONE"
	picture = "NONE"
	is_triggered_only = yes
	hidden = yes

	trigger = {
		has_global_flag = ai_building_scripts_on
		NOT = { has_country_modifier = ai_build_buildings_cooldown }
		OR = {
			is_great_power = yes
			treasury = 6000
		}
		is_in_deficit = no
		is_at_war = no
		treasury = 2000
		NOT = { num_of_loans = 1 }
		NOT = { primitives = yes }
		NOT = { government = native }
	}

	immediate = {
		while = {
			limit = {
				treasury = 1000
				any_owned_province = {
					num_free_building_slots = 1
					not_under_construction = yes
					is_state = yes
				}
			}
			random_owned_province = {
				limit = {
					num_free_building_slots = 1
					not_under_construction = yes
					is_state = yes
					NOT = { has_townhall_building = yes }
				}
				ai_build_townhall_effect = yes
			}
		}
		while = {
			limit = {
				treasury = 1000
				any_owned_province = {
					num_free_building_slots = 1
					not_under_construction = yes
					is_state = yes
					OR = {
						trade_goods = grain
						trade_goods = fruits
						trade_goods = rice
					}
					NOT = { has_granary_building = yes }
				}
			}
			random_owned_province = {
				limit = {
					num_free_building_slots = 1
					not_under_construction = yes
					is_state = yes
					OR = {
						trade_goods = grain
						trade_goods = fruits
						trade_goods = rice
					}
					NOT = { has_granary_building = yes }
				}
				ai_build_granary_effect = yes
			}
		}
		while = {
			limit = {
				treasury = 1000
				any_owned_province = {
					num_free_building_slots = 1
					not_under_construction = yes
					is_state = yes
					OR = {
						base_tax = 2
						base_production = 4
					}
					NOT = { has_production_building = yes }
				}
			}
			random_owned_province = {
				limit = {
					num_free_building_slots = 1
					not_under_construction = yes
					is_state = yes
					OR = {
						base_tax = 2
						base_production = 4
					}
					NOT = { has_production_building = yes }
				}
				ai_build_production_effect = yes
			}
		}
		while = {
			limit = {
				treasury = 1000
				any_owned_province = {
					NOT = { num_free_building_slots = 1 }
					not_under_construction = yes
					is_state = yes
					NOT = { has_province_modifier = land_reclamation_5 }
				}
			}
			random_owned_province = {
				limit = {
					NOT = { num_free_building_slots = 1 }
					not_under_construction = yes
					is_state = yes
					NOT = { has_province_modifier = land_reclamation_5 }
				}
				ai_build_reclaimation_effect = yes
			}
		}
	}

	after = {
		add_country_modifier = {
			name = ai_build_buildings_cooldown
			duration = 412
		}
	}
	option = {}
}

country_event = { #great conquerors
	id = ai_handling.6
	title = "NONE"
	desc = "NONE"
	picture = "NONE"
	hidden = yes
	trigger = {
		REB = {
			NOT = {
				has_country_modifier = ai_search_conqueror_cooldown
			}
		}
		is_part_of_hre = no
		OR = {
			has_global_flag = ai_conquerors_active_easy
			has_global_flag = ai_conquerors_active_medium
			has_global_flag = ai_conquerors_active_hard
		}
		NOT = {
			calc_true_if = {
				all_country = {
					has_ruler_flag = ai_the_ai_is_conqueror
				}
				amount = 3
			}
		}
	}
	mean_time_to_happen = {
		months = 1000
		modifier = {
			factor = 2
			calc_true_if = {
				all_country = {
					has_ruler_flag = ai_the_ai_is_conqueror
				}
				amount = 1
			}
		}
		modifier = {
			factor = 3
			calc_true_if = {
				all_country = {
					has_ruler_flag = ai_the_ai_is_conqueror
				}
				amount = 2
			}
		}
	}
	immediate = {
		if = {
			limit = {
				NOT = {
					calc_true_if = {
						all_country = {
							has_ruler_flag = ai_the_ai_is_conqueror
						}
						amount = 3
					}
				}
			}
			REB = {
				add_country_modifier = {
					name = ai_search_conqueror_cooldown
					duration = 3653
				}
			}
			random_country = {
				limit = {
					NOT = {
						has_ruler_flag = ai_the_ai_is_conqueror
					}
					NOT = {
						has_country_modifier = ai_conqueror_cooldown
					}
					OR = {
						government = monarchy
						uses_horde_unity = yes
					}
					ai = yes
					ruler_age = 40
					total_development = 75
					num_of_owned_areas = 5
					is_subject = no
					is_native_tribe = no
					is_part_of_hre = no
					OR = {
						AND = {
							mil = 3
							dip = 3
							adm = 3
						}
						AND = {
							mil = 5
							OR = {
								ruler_has_personality = conqueror_personality
								ruler_has_personality = the_great_personality
								ruler_has_personality = inspiring_leader_personality
							}
						}
					}
				}
				set_ruler_flag = ai_the_ai_is_conqueror
				remove_ruler_personality = ai_personality
				add_ruler_personality = ai_personality_conqueror
				add_country_modifier = {
					name = ai_conqueror_cooldown
					duration = 18262
				}
				capital_scope = {
					add_unit_construction = {
						type = infantry
						amount = 10
						speed = 0
						cost = 0
					}
					add_unit_construction = {
						type = cavalry
						amount = 5
						speed = 0
						cost = 0
					}
				}
				add_manpower = 15
				add_treasury = 200
				add_yearly_manpower = 5
				add_years_of_income = 3
				create_general = {
					tradition = 100
				}
				create_general = {
					tradition = 50
				}
			}
		}
		else = {
		}
	}
	after = {
		if = {
			limit = {
				has_ruler_flag = ai_the_ai_is_conqueror
				is_at_war = no
			}
			if = {
				limit = {
					uses_horde_unity = yes
				}
				capital_scope = {
					build_to_forcelimit = {
						infantry = 0.4
						cavalry = 0.5
						artillery = 0.1
					}
				}
			}
			if = {
				limit = {
					uses_horde_unity = no
				}
				capital_scope = {
					build_to_forcelimit = {
						infantry = 0.5
						cavalry = 0.2
						artillery = 0.3
					}
				}
			}
			add_yearly_manpower = 10
			add_adm_power = 150
			add_dip_power = 150
			add_mil_power = 150
		}
		else_if = {
			limit = {
				has_ruler_flag = ai_the_ai_is_conqueror
				is_at_war = yes
			}
			add_yearly_manpower = 10
			add_adm_power = 75
			add_dip_power = 75
			add_mil_power = 75
		}
		else = {
		}
		every_country = {
			limit = {
				has_ruler_flag = ai_the_ai_is_conqueror
				NOT = {
					has_country_flag = ai_had_conqueror
				}
			}
			set_country_flag = ai_had_conqueror
		}
	}
	option = {
	}
}
