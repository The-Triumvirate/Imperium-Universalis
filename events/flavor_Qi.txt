		months = 1
	}
	
	option = {
		name = "flavor_Qi.1.a"

			12.5 = { add_ruler_modifier = { name = qi_bashen_tianzhu duration = -1 } }

			12.5 = { add_ruler_modifier = { name = qi_bashen_bingzhu duration = -1 } }
			12.5 = { add_ruler_modifier = { name = qi_bashen_yinzhu duration = -1 } }

		}
	}
