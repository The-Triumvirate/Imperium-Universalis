country_event = {
	id = religion_china.1
	title = "religion_china.1.t"
	desc = "religion_china.1.d"
	picture = ruler
	major = yes

	trigger = {
		NOT = { is_religion_enabled = confucianism }
		culture_group = chinese_group
		religion_group = chinese_religions
		OR = {
			is_year = 254
			AND = {
				is_year = 229
				has_global_flag = zhanguokaishi
			}
		}
	}

	mean_time_to_happen = {
		months = 60
		modifier = {
			factor = 10
			current_age = age_of_empires
		}
		modifier = {
			factor = 0.5
			tag = LUU
		}
		modifier = {
			factor = 0.5
			has_global_flag = zhanguokaishi
		}
	}

	option = {
		name = "religion_china.1.a"
		define_advisor = {
			type = scholar_advisor
			name = "Confucius"
			skill = 3
			discount = yes
		}
		add_adm_power = 100
		add_dip_power = 100
		add_mil_power = 100
		enable_religion = confucianism
		#add_reform_desire = -0.5
	}
}

########################################################################################
# shenism spreads
########################################################################################
province_event = {
	id = religion_china.5
	title = "religion_china.5.t"
	desc = "religion_china.5.d"
	picture = religious

	trigger = {
		religion_group = eastern_religions
		NOT = {
			religion = tengri
		}
		OR = {
			any_neighbor_province = { religion_group = chinese_religions }
			culture_group = chinese_group
		}
		owner = { religion_group = chinese_religions }
		NOT = { has_province_modifier = qig_wuism_not_converting }
	}

	mean_time_to_happen = {
		months = 300
		modifier = {
			factor = 0.15
			culture_group = chinese_group
		}
		modifier = {
			factor = 1.25
			culture = jingman
		}
		modifier = {
			factor = 3
			owner = { NOT = { culture_group = chinese_group } }
			NOT = { culture_group = chinese_group }
		}
		modifier = {
			factor = 1.25
			owner = { NOT = { adm_tech = 4 } }
		}
		modifier = {
			factor = 0.8
			owner = { adm_tech = 9 }
		}
	}

	option = {
		name = "religion_china.5.a"
		change_religion = owner
	}
}
