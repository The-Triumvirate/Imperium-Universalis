country_event = {
	id = china.238
	title = "china.238.t"
	desc = "china.238.d"
	picture = deligation
	major = yes
	
	trigger = {
		NOT = { exists = ZHO }
		4349 = {
			owner = {
				NOT = {
					religion = chinese_religion
				}
			}
		}
		is_emperor = yes
		religion = chinese_religion
		num_of_cities = 8
	}
	
	mean_time_to_happen = {
		months = 6
	}

	option = {
		name = "china.238.a"
		hidden_effect = {
			every_province = {
				limit = {
					has_province_flag = zhou_fled_here_pf
				}
				clr_province_flag = zhou_fled_here_pf
			}
		}
		ROOT = {
			save_global_event_target_as = zhou_savior_country
		}
		random_owned_province = {
			cede_province = ZHO
			add_core = ZHO
			remove_core = ROOT
			set_province_flag = zhou_fled_here_pf
		}
		create_alliance = ZHO
		add_historical_friend = ZHO
		ZHO = {
			add_historical_friend = ROOT
			set_dynasty = "Ji"
			change_religion = chinese_religion
			country_event = {
				id = chinese_school_events.1
				days = 1
			}
		}
		add_trust = {
			who = ZHO
			value = 100
			mutual = yes
		}
		add_favors = {
		   who = ZHO
		   amount = 100
		}
		ZHO = {
			set_country_flag = zhou_had_to_flee_flag
		}
		4349 = {
			owner = {
				country_event = { id = china.241 }
			}
		}
	}

		name = "china.238.b"

				}

					save_global_event_target_as = zhou_savior_country
				}
				random_owned_province = {
					cede_province = ZHO
					add_core = ZHO
					remove_core = PREV
					set_province_flag = zhou_fled_here_pf
				}
				create_alliance = ZHO
				add_historical_friend = ZHO
				ZHO = {
					add_historical_friend = THIS
					set_dynasty = "Ji"
					change_religion = chinese_religion
					country_event = {
						id = chinese_school_events.1
						days = 1
					}
				}
				add_trust = {
					who = ZHO

					mutual = yes
				}
				add_favors = {
				   who = ZHO

				}
