#BUILD ARMY INITIALIZATION
ai_build_to_force_limit = {
	if = {
		limit = { is_emperor = no }
		random_owned_province = {
			limit = { highest_supply_limit_in_area = yes }
			build_to_forcelimit = {
				infantry = 0.6
				cavalry = 0.1
				artillery = 0.2
			}
		}
		if = {
			limit = { num_of_ports = 1 }
			random_owned_province = {
				limit = { has_port = yes }
				build_to_forcelimit = {
					heavy_ship = 0.2
					light_ship = 0.1
					galley = 0.4
					transport = 0.3
				}
			}
		}
	}
	else = {
		random_owned_province = {
			limit = { highest_supply_limit_in_area = yes }
			build_to_forcelimit = {
				infantry = 0.5
				cavalry = 0.1
				artillery = 0.2
			}
		}
		if = {
			limit = { num_of_ports = 1 }
			random_owned_province = {
				limit = { has_port = yes }
				build_to_forcelimit = {
					heavy_ship = 0.2
					light_ship = 0.1
					galley = 0.4
					transport = 0.3
				}
			}
		}
	}
}
#SET CROWNLAND INITIALIZATION
ai_set_crown_land = {
	if = {
		limit = { NOT = { crown_land_share = 25 } }
		change_estate_land_share = {
			estate = all
			share = -10
		}
	}
	add_estate_loyalty = {
		estate = all
		loyalty = 15
		short = yes
	}
}

	}
}

ai_alliance_initialization_effect = {
	#HISTORICAL FRIEND = ALLIANCE
	if = {
		limit = {
			num_of_free_diplomatic_relations = 1
			NOT = { tag = ZHO }
			any_known_country = {
				historical_friend_with = ROOT
				NOT = { tag = ZHO }
				num_of_free_diplomatic_relations = 1
				ai = yes
				ai_is_valid_for_alliance_trigger = yes
			}
		}
		every_known_country = {
			limit = {
				historical_friend_with = ROOT
				num_of_free_diplomatic_relations = 1
				ai = yes
				ai_is_valid_for_alliance_trigger = yes
			}
			create_alliance = ROOT
		}
	}

	#GUARANTEED = ALLIANCE
	if = {
		limit = {
			any_known_country = {
				is_subject = no
				guaranteed_by = ROOT
				ai = yes
			}
		}
		every_known_country = {
			limit = {
				is_subject = no
				guaranteed_by = ROOT
				ai = yes
			}
			create_alliance = ROOT
		}
	}
	#EMPEROR - ELECTOR = ALLIANCE
	#if = {

	#		}
	#	}
	#	every_elector = {
	#		limit = {
	#			total_development = 35
	#			num_of_free_diplomatic_relations = 1
	#			ai = yes
	#			ai_is_valid_for_alliance_trigger = yes
	#		}
	#		create_alliance = ROOT
	#	}
	#}
	#GREAT POWER - NOT WEAK NATION = ALLIANCE
	if = {
		limit = {
			is_great_power = yes
			num_of_free_diplomatic_relations = 1
			any_known_country = {
				army_strength = {
					who = ROOT
					value = 0.45
				}
				NOT = {
					border_distance = {
						who = ROOT
						distance = 175
					}
				}
				border_distance = {
					who = ROOT
					distance = 50
				}
				religion_group = ROOT
				num_of_free_diplomatic_relations = 1
				ai = yes
				ai_is_valid_for_alliance_trigger = yes
			}
		}
		random_known_country = {
			limit = {
				army_strength = {
					who = ROOT
					value = 0.45
				}
				NOT = {
					border_distance = {
						who = ROOT
						distance = 175
					}
				}
				border_distance = {
					who = ROOT
					distance = 55
				}
				NOT = { is_emperor = yes }
				NOT = { is_elector = yes }
				religion_group = ROOT
				num_of_free_diplomatic_relations = 1
				ai = yes
				ai_is_valid_for_alliance_trigger = yes
			}
			create_alliance = ROOT
		}
	}
	if = {
		limit = {
			OR = {
				is_great_power = yes
				total_development = 165
			}
		}
		every_neighbor_country = {
			limit = {
				ai = yes
				NOT = { total_development = 105 }
				is_part_of_hre = no
				NOT = { religion = wuism }
				NOT = { is_subject_of = ROOT }
				NOT = { alliance_with = ROOT }
				NOT = { historical_friend_with = ROOT }
				#NOT = { tag = ZHO }
			}
			if = {
				limit = {
					capital_scope = {
						superregion = china_superregion
					}
				}
				#add_opinion = {
				#	who = ROOT
				#	modifier = ai_conq_box_lesser
				#}
				#reverse_add_opinion = {
				#	who = ROOT
				#	modifier = ai_conq_box_lesser
				#}
			}
			else = {
				add_opinion = {
					who = ROOT
					modifier = ai_conq_box
				}
				reverse_add_opinion = {
					who = ROOT
					modifier = ai_conq_box
				}
			}
		}
	}
}

ai_build_forts_effect = {
	if = {
		limit = {
			can_build = fortress_building
			PREV = {
				treasury = 700
				monthly_income = 30
				current_income_balance = 10
			}
		}
		add_building_construction = {
			building = fortress_building
			speed = 1
			cost = 0.75
		}
	}
	else_if = {
		limit = {
			can_build = large_city_walls
			PREV = {
				treasury = 600
				monthly_income = 25
				current_income_balance = 8
			}
		}
		add_building_construction = {
			building = large_city_walls
			speed = 1
			cost = 0.75
		}
	}

		limit = {
			can_build = small_city_walls
			PREV = {
				treasury = 400

			}
		}
		add_building_construction = {
			building = small_city_walls
			speed = 1

		}
	}

			can_build = border_garrison

}
