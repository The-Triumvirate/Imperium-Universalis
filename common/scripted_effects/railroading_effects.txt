ai_rome_forum_mission_effect = {
	if = {
		limit = {
			mission_completed = rom_the_roman_forum_mission
			NOT = { mission_completed = rom_construction_of_the_servian_wall_mission }
			NOT = { has_country_flag = rom_ai_serv_wall_mission }
			mil_tech = 2
		}
		53 = { add_building = small_city_walls }
		add_treasury = 100
		set_country_flag = rom_ai_serv_wall_mission
	}
}

ai_rome_conquer_first_colonia_effect = {
	if = {
		limit = {
			mission_completed = rom_more_than_just_a_city_mission
			NOT = { mission_completed = rom_the_first_colonia_mission }
			NOT = { has_country_flag = rom_ai_first_colonia_mission }
		}
		if = {
			limit = { 5642 = { NOT = { is_core = ROM } } }
			add_territorial_core = 5642
			owner = { break_alliance_insult_effect = { WHO = ROM } }
		}
		if = {
			limit = { 52 = { NOT = { is_core = ROM } } }
			add_territorial_core = 52
			owner = { break_alliance_insult_effect = { WHO = ROM } }
		}
		if = {
			limit = { 5638 = { NOT = { is_core = ROM } } }
			add_territorial_core = 5638
			owner = { break_alliance_insult_effect = { WHO = ROM } }
		}
		set_country_flag = rom_ai_first_colonia_mission
	}
}

ai_rome_build_first_colonia_effect = {
	if = {
		limit = {
			mission_completed = rom_more_than_just_a_city_mission
			NOT = { mission_completed = rom_the_first_colonia_mission }
			NOT = { has_country_flag = rom_ai_first_colonia_built_mission }
			owns = 5642
			owns = 52
			owns = 5638
		}
		52 = {
			if = {
				limit = {
					NOT = { has_road_building = yes }
				}
				add_building = paved_road
			}
			if = {
				limit = {
					NOT = { has_shipyard_building = yes }
				}
				add_building = small_shipyard
			}
			if = {
				limit = {
					NOT = { has_temple_building = yes }
				}
				add_building = small_temple
			}
		}
		5642 = {
			if = {
				limit = {
					NOT = { has_road_building = yes }
				}
				add_building = paved_road
			}
			if = {
				limit = {
					NOT = { has_army_building = yes }
				}
				add_building = small_army_camp
			}
		}
		5638 = {
			if = {
				limit = {
					NOT = { has_road_building = yes }
				}
				add_building = paved_road
			}
			if = {
				limit = {
					NOT = { has_army_building = yes }
				}
				add_building = small_army_camp
			}
		}
		set_country_flag = rom_ai_first_colonia_built_mission
	}
}
