estate_handler_for_ai = {
	if = {
		limit = { #so ai doesn't deathloop at small landshare
			NOT = { crown_land_share = 20 }
		}
		take_estate_land_share_small = { estate = estate_priesthood }
		take_estate_land_share_small = { estate = estate_nobility }
		take_estate_land_share_small = { estate = estate_merchants }
		take_estate_land_share_medium = { estate = estate_clans }
	}
}
handle_parliament_seats_effect = {
	set_variable = {
		which = current_parliament_seat_variable
		value = 0
	}
	every_owned_province = {
		limit = { has_seat_in_parliament = yes }
		PREV = {
			change_variable = {
				which = current_parliament_seat_variable
				value = 1
			}
		}
	}
	export_to_variable = {
		which = max_parliament_seat_variable
		value = modifier:max_possible_parliament_seats
	}
	if = {
		limit = {
			check_variable = {
				which = max_parliament_seat_variable
				which = current_parliament_seat_variable
			}
		}
		set_country_flag = can_add_p_seats
	}
	else = {
		clr_country_flag = can_add_p_seats
	}
}

player_scripted_effect_monthly = { #happens every month for player
	if = { #within if = happens every 12 months
		limit = {
			check_variable = {
				which = monthly_tick_player_check
				value = 11
			}
		}
		set_variable = {
			which = monthly_tick_player_check
			value = 0
		}
		every_owned_province = {
			SD_update_provslots = yes
			SD_update_prov_output = yes
		}
		if = {
			limit = {
				NOT = { has_country_flag = hidden_slaves_event }
			}
			country_event = {
				id = slaves.1
			}
		}
		else = {
			country_event = {
				id = slaves.101
			}
		}
		handle_parliament_seats_effect = yes
	}
	else = {
		change_variable = {
			which = monthly_tick_player_check
			value = 1
		}
	}
}
