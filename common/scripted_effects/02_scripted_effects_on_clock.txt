estate_handler_for_ai = {
	if = {
		limit = { #so ai doesn't deathloop at small landshare
			NOT = { crown_land_share = 20 }
		}
		take_estate_land_share_small = { estate = estate_priesthood }
		take_estate_land_share_small = { estate = estate_nobility }
		take_estate_land_share_small = { estate = estate_merchants }
		take_estate_land_share_medium = { estate = estate_clans }
	}
}


player_scripted_effect_monthly = { #happens every month for player
}

ai_scripted_effect_monthly = { #happens every month for player subjects that are ai
	estate_handler_for_ai = yes
}

ai_scripted_effect_bi_yearly = { #happens every 2 years for ai
	estate_handler_for_ai = yes
}
