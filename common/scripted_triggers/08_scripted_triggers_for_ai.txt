#DIPLOMACY
ai_is_valid_for_alliance_trigger = {
	is_free_or_tributary_trigger = yes
	OR = {
		ai_attitude = {
			who = ROOT
			attitude = attitude_neutral
		}
		ai_attitude = {
			who = ROOT
			attitude = attitude_friendly
		}
	}
	has_opinion = {
		who = ROOT
		value = 25
	}
	NOT = {
		OR = {
			alliance_with = ROOT
			is_rival = ROOT
		}
	}
}

			is_subject_of = ROOT
			overlord_of = ROOT
			alliance_with = ROOT
			truce_with = ROOT

				who = ROOT
				value = 50
			}
			AND = {
				tag = ZHO

					is_emperor = yes
				}
			}
			AND = {
				is_emperor = yes

					tag = ZHO
				}
			}
		}
	}
}

ai_fort_dev_requirements_trigger = {
	OR = {
		if = {
			limit = { has_terrain = mountain }
			development = 4
		}
		else_if = {
			limit = { has_terrain = marsh }
			development = 4
		}
		else_if = {
			limit = { high_terrain_not_mountain_trigger = yes }
			development = 5
		}
		else_if = {
			limit = { forest_terrain_trigger = yes }
			development = 8
		}
		else_if = {
			limit = { flat_terrain_trigger = yes }
			development = 6
		}
		else = {
			development = 9
		}
	}
}

ai_can_build_fort = {
	if = {
        limit = {
			mil_tech = 22
            treasury = 800
            monthly_income = 25
            current_income_balance = 8
        }
    }
    else_if = {
        limit = {
			mil_tech = 13
            treasury = 600
            monthly_income = 20
            current_income_balance = 6
        }
    }
    else_if = {
        limit = {
			mil_tech = 2
            treasury = 400
            monthly_income = 15
            current_income_balance = 5
        }
    }
    else_if = {
        limit = {
			mil_tech = 1
            treasury = 225
            monthly_income = 10
            current_income_balance = 4
        }
    }
	else = {
		always = no
	}
}
