	name = invite_scholar_buttonNEW
	tooltip = invite_scholar_buttonNEW_tt
	potential = { religion_group = chinese_religions }
    trigger = {
		ai = no
		stability = 1
		dip_power = 50
		government_reform_progress = 20
		custom_trigger_tooltip = {
			tooltip = NOT_ALREADY_HAVE_SCHOLAR_CHINA
			OR = {
				NOT = {
					has_country_flag = china_currently_invited_scholar
				}
				had_country_flag = {
					flag = china_currently_invited_scholar
					days = 7300
				}
			}
		}
		hidden_trigger = { NOT = { has_country_flag = selecting_new_school_evn_china } }
	}
	effect = {
		clr_country_flag = china_currently_invited_scholar
		set_country_flag = selecting_new_school_evn_china
		add_dip_power = -50
		change_government_reform_progress = -20
		country_event = { id = chinese_school_events.2 }
	}
}

	name = join_heretic_league_button
	tooltip = join_heretic_league_button_tt
	potential = { hre_leagues_enabled = yes }
    trigger = {
		is_at_war = no
		is_subject = no
		is_emperor = no
		NOT = {
			in_league = chinese_religion
		}
		custom_trigger_tooltip = {
			tooltip = NO_TRUCE_WITH_EMPEROR
			emperor = {
				NOT = {
					truce_with = ROOT
				}
			}
		}
		custom_trigger_tooltip = {
			tooltip = LEAGUES_WAR_NOT_ONGOING
			emperor = {
				is_in_league_war = no
			}
		}
		custom_trigger_tooltip = {
			tooltip = CANNOT_JOIN_LEAGUE_TIMEOUT
			OR = {
				NOT = { has_country_flag = cant_join_league_5_years }
				had_country_flag = {
					flag = cant_join_league_5_years
					days = 1825
				}
			}
			OR = {
				NOT = { has_country_flag = cant_join_league_10_years }
				had_country_flag = {
					flag = cant_join_league_10_years
					days = 3650
				}
			}
		}
	}
	frame = {
		number = 2
		trigger = { in_league = liyuebadao }
	}
	frame = {
		number = 1
		trigger = { always = yes }
	}
