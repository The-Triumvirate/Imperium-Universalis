#zhou may grant use of palace guards
zhou_grant_guards = {
	category = papacy
	alert_index = 11
	#alert_tooltip = justify_invasion_alert_tooltip.1

	require_acceptance = no
	is_visible = {
		tag = ZHO
		religion = chinese_religion
		NOT = { war_with = FROM }
		NOT = { tag = ZHG }
		FROM = {
			religion = chinese_religion
			is_subject = no
			NOT = {
				has_country_flag = allowed_to_use_zhou_guards
			}
			is_excommunicated = no
		}
		OR = {
			NOT = { has_country_flag = zhou_gave_out_guards }
			had_country_flag = {
				flag = zhou_gave_out_guards
				days = 1825
			}
		}
	}
	is_allowed = {
		is_at_war = no
	}
	on_accept = {
		clr_country_flag = zhou_gave_out_guards
		set_country_flag = zhou_gave_out_guards
		FROM = { country_event = { id = china.209 } }
	}

	ai_will_do = {
		FROM = {
			OR = {
				is_emperor = yes
				is_papal_controller = yes
				has_country_modifier = zhou_papacy_tribute_modifier
			}
			total_development = 50
			is_subject = no
			NOT = { war_score = -20 }
			is_in_war = {
				duration = 180
			}
		}
	}
}
