
bazhu_move_capital = {
	category = emperor
	#alert_index = 11
	#alert_tooltip = china_appoint_shenshou_alert_tooltip.1

	require_acceptance = no
	
	is_visible = {
		hre_reform_passed = emperor_reichskrieg
		is_emperor = yes
		FROM = {
			is_subject_of_type = fengjian_subject
			num_of_cities = 2
		}
	}
	is_allowed = {
		hidden_trigger = { hre_reform_passed = emperor_reichskrieg }
		is_at_war = no
		imperial_influence = 5
	}
	on_accept = {
		add_imperial_influence = -5
		FROM = {
			random_owned_province = {
				limit = {
					is_capital = no
				}
				FROM = {
					set_capital = PREV
				}
			}
		}
	}

	ai_will_do = {
		always = no
	}
}
