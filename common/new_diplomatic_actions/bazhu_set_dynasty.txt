
bazhu_set_dynasty = {
	category = emperor
	#alert_index = 11
	#alert_tooltip = china_appoint_shenshou_alert_tooltip.1

	require_acceptance = no
	
	is_visible = {
		hre_reform_passed = emperor_reichskrieg
		FROM = {
			is_subject_of_type = fengjian_subject
			NOT = {
				tag = ZHO
			}
		}
		is_emperor = yes
		NOT = {
			has_country_flag = bazhu_selecting_dynasty_rn
		}
	}
	is_allowed = {
		hidden_trigger = { hre_reform_passed = emperor_reichskrieg }
		is_at_war = no
		OR = {
			AND = {
				FROM = {
					NOT = {
						total_development = 25
					}
				}
				imperial_influence = 2
			}
			AND = {
				FROM = {
					total_development = 25
					NOT = {
						total_development = 75
					}
				}
				imperial_influence = 5
			}
			AND = {
				FROM = {
					total_development = 75
					NOT = {
						total_development = 200
					}
				}
				imperial_influence = 10
			}
			AND = {
				FROM = {
					total_development = 200
				}
				imperial_influence = 15
			}
		}
	}
	on_accept = {
		FROM = {
			save_global_event_target_as = bazhu_dynasty_change_target
		}
		set_country_flag = bazhu_selecting_dynasty_rn
		country_event = {
			id = china.221
		}
	}

	ai_will_do = {
		always = no
	}
}
