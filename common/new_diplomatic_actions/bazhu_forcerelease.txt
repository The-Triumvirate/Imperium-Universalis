#allows bazhu to make zhuxia state to release all countries
bazhu_forcerelease = {
	category = emperor
	#alert_index = 11
	#alert_tooltip = justify_invasion_alert_tooltip.1

	require_acceptance = no
	is_visible = {
		hre_reform_passed = emperor_reichskrieg
		religion = chinese_religion
		FROM = {
			is_part_of_hre = yes
			NOT = {
				tag = ZHO
			}
		}
	}
	is_allowed = {
		hidden_trigger = { hre_reform_passed = emperor_reichskrieg }
		is_subject = no
		is_at_war = no
		OR = {
			AND = {
				FROM = {
					NOT = {
						total_development = 150
					}
				}
				imperial_influence = 10
			}
			AND = {
				FROM = {
					total_development = 150
					NOT = {
						total_development = 300
					}
				}
				imperial_influence = 25
			}
			AND = {
				FROM = {
					total_development = 300
					NOT = {
						total_development = 500
					}
				}
				imperial_influence = 50
			}
			AND = {
				FROM = {
					total_development = 500
				}
				imperial_influence = 80
			}
		}
	}
	on_accept = {
		hidden_effect = {
			if = {
				limit = {
					NOT = {
						total_development = 150
					}
				}
				add_imperial_influence = -10
			}
			else_if = {
				limit = {
					total_development = 150
					NOT = {
						total_development = 300
					}
				}
				add_imperial_influence = -25
			}
			else_if = {
				limit = {
					total_development = 300
					NOT = {
						total_development = 500
					}
				}
				add_imperial_influence = -50
			}
			else_if = {
				limit = {
					total_development = 500
				}
				add_imperial_influence = -80
			}
		}
		FROM = {
			release_all_possible_countries = yes
			release_all_subjects = yes
		}
		#recalc fengjian for new released states
		every_country = {
			limit = {
				OR = {
					is_part_of_hre = yes
					tag = XNC
					tag = HUA
					tag = KUI
				}
				is_emperor = no
				num_of_cities = 1
				is_subject = no
				#NOT = { tag = ZHO }
				NOT = { tag = ZHG }
			}
			country_event = { id = china.89 }
		}
	}

	ai_will_do = {
		always = no
	}
}
