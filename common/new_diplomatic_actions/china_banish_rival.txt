#Allows banishment of rival nations (base-game function is hardly useable because AI always buys indulgence)
china_banish_rival = {
	category = papacy
	#alert_index = 11
	#alert_tooltip = justify_invasion_alert_tooltip.1

	require_acceptance = no
	is_visible = {
		is_papal_controller = yes
		NOT = { FROM = { is_subject_of = ROOT } }
		religion = chinese_religion
		FROM = {
			religion = chinese_religion
			is_excommunicated = no
		}
	}
	is_allowed = {
		hidden_trigger = { is_papal_controller = yes }
		is_subject = no
		is_at_war = no
		all_known_country = {
			is_excommunicated = no
		}
		OR = {
			FROM = {
				AND = {
					is_enemy = ZHO #intentional, if it was a rival of the qingshi it'd be too op
					is_subject = no
				}
			}
			mission_completed = zho_2_alt_neo_reforms
		}
	}
	on_accept = {
		excommunicate = FROM
		FROM = {
			country_event = { id = china.29 }
		}
	}

	ai_will_do = {
		NOT = {
			has_opinion = {
				who = FROM
				value = -125
			}
		}
		FROM = {
			months_of_ruling = 60
			is_subject = no
			NOT = {
				personality = ai_diplomat
			}
		}
	}
}
